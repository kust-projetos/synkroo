**src/services/custom-fields/types.ts:**
```typescript
export type FieldType = 'text' | 'number' | 'date' | 'select' | 'checkbox';

export interface CustomFieldDefinition {
  id: string;
  clinic_id: string;
  name: string;
  field_type: FieldType;
  options: Array<{ label: string; value: string }>; // Only for 'select' type
  required: boolean;
  sort_order: number;
  is_active: boolean;
  created_at: string;
  updated_at: string;
}

export interface CustomFieldValue {
  id: string;
  definition_id: string;
  contact_id: string;
  contact_type: 'patient' | 'lead';
  clinic_id: string;
  value_text: string | null;
  value_number: number | null;
  value_date: string | null;
  value_boolean: boolean | null;
  value_json: unknown | null;
  // Computed from definition
  field_name?: string;
  field_type?: FieldType;
}

export interface FieldValueInput {
  definition_id: string;
  value: unknown; // Type coerced based on field_type
}

export interface FieldDefinitionCreateInput {
  name: string;
  field_type: FieldType;
  options?: Array<{ label: string; value: string }>;
  required?: boolean;
  sort_order?: number;
}

export interface FieldDefinitionUpdateInput {
  name?: string;
  options?: Array<{ label: string; value: string }>;
  required?: boolean;
  sort_order?: number;
  is_active?: boolean;
}

export interface FieldDefinitionExport {
  version: 1;
  exported_at: string;
  definitions: Array<{
    name: string;
    field_type: FieldType;
    options: Array<{ label: string; value: string }>;
    required: boolean;
    sort_order: number;
  }>;
}
```

**src/services/custom-fields/definitions.service.ts:**
All functions take `clinicId: string` as first param.

1. `getDefinitions(clinicId, { activeOnly = false } = {})`: Query custom_field_definitions WHERE clinic_id, optional filter is_active=true, order by sort_order. Return CustomFieldDefinition[].

2. `createDefinition(clinicId, input: FieldDefinitionCreateInput)`: Validate name unique per clinic (Supabase UNIQUE constraint handles this). For 'select' type, validate options is non-empty array. Insert and return.

3. `updateDefinition(clinicId, definitionId, input: FieldDefinitionUpdateInput)`: Update WHERE id AND clinic_id. Return updated.

4. `deleteDefinition(clinicId, definitionId)`: Soft delete (set is_active=false) rather than hard delete, preserving existing values. Return void.

5. `exportDefinitions(clinicId)`: Get all active definitions, return FieldDefinitionExport JSON.

6. `importDefinitions(clinicId, data: FieldDefinitionExport)`: For each definition in data.definitions, check if name already exists for clinic. If not, create it. Skip existing. Return count of created + skipped.

**src/services/custom-fields/values.service.ts:**

1. `getValuesForContact(clinicId, contactId, contactType)`: Query custom_field_values WHERE contact_id AND contact_type AND clinic_id. Join with custom_field_definitions to include field_name and field_type. Return CustomFieldValue[].

2. `upsertValues(clinicId, contactId, contactType, values: FieldValueInput[])`: For each value input:
   - Look up the definition to get field_type
   - Route to the correct typed column based on field_type:
     - text -> value_text
     - number -> value_number
     - date -> value_date
     - checkbox -> value_boolean
     - select -> value_json (for single select) or value_json (for multi)
   - Build upsert rows with all typed columns (set non-matching to null)
   - Use Supabase `.upsert(rows, { onConflict: 'definition_id,contact_id,contact_type' })`

3. `searchByCustomField(clinicId, definitionId, value, fieldType)`: Query custom_field_values WHERE definition_id AND clinic_id AND the appropriate typed column:
   - text: `value_text ILIKE '%value%'`
   - number: `value_number eq value`
   - date: `value_date eq value`
   - boolean: `value_boolean eq value`
   - select: `value_json @> value`
   Return array of { contact_id, contact_type, value_* } for use as filter by the contacts service.

export function useCustomFieldDefinitions(activeOnly = true) {
  return useQuery({
    queryKey: [queryKeys.customFieldDefinitions, activeOnly],
    queryFn: async () => { const res = await fetch(`/api/custom-fields/definitions?active_only=${activeOnly}`); if (!res.ok) throw new Error('Failed'); return res.json(); }
  })
}

export function useCustomFieldValues(contactId: string, contactType: string) {
  return useQuery({
    queryKey: [queryKeys.customFieldValues, contactId, contactType],
    queryFn: async () => { const res = await fetch(`/api/custom-fields/values?contact_id=${contactId}&contact_type=${contactType}`); if (!res.ok) throw new Error('Failed'); return res.json(); },
    enabled: !!contactId && !!contactType,
  })
}
